cmake_minimum_required(VERSION 3.13)
project(downloadTool LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)

option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
add_library(download_options INTERFACE)
if(ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
    target_compile_options(download_options INTERFACE
        -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(download_options INTERFACE -fsanitize=address,undefined)
endif()

add_library(download_base STATIC Download.cpp)
target_include_directories(download_base PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(download_base PUBLIC CURL::libcurl Threads::Threads download_options)
add_executable(downloadTool main.cpp HttpDownload.cpp)
target_link_libraries(downloadTool PRIVATE download_base)

include(CTest)
if(BUILD_TESTING)
    add_executable(test-write-data tests/test_write_data.cpp)
    target_link_libraries(test-write-data PRIVATE download_base)
    foreach(case bytes boundary overrun invalid empty seek-error write-error concurrent)
        add_test(NAME write-${case} COMMAND test-write-data ${case})
        set_tests_properties(write-${case} PROPERTIES TIMEOUT 15
            FAIL_REGULAR_EXPRESSION "AddressSanitizer;UndefinedBehaviorSanitizer;runtime error:;LeakSanitizer")
    endforeach()
endif()
